OfflineTransport.Route = request =>
{
    Check(request.Headers.Authorization is null, "no token sent");
    if (request.RequestUri!.Host == "api.github.com")
    {
        Check(request.RequestUri.AbsolutePath == "/repos/fixture/updater/releases/latest", "exact -10 API latest path");
        if (apiStatus != HttpStatusCode.OK) return new(apiStatus);
        var assets = new List<object>
        {
            new { name = sc, browser_download_url = prefix + sc, size = bytes.Length, digest = withDigest ? "sha256:" + hash : null },
            new { name = fdd, browser_download_url = prefix + fdd, size = bytes.Length, digest = withDigest ? "sha256:" + hash : null },
            new { name = "update.json", browser_download_url = prefix + "update.json", size = 0, digest = (string?)null }
        };
        if (!missingChecksum) assets.Add(new { name = "SHA256SUMS", browser_download_url = prefix + "SHA256SUMS", size = 0, digest = (string?)null });
        if (lastMatch) assets.Add(new { name = "MikuN2N-extra-win-x64.zip", browser_download_url = prefix + "MikuN2N-extra-win-x64.zip", size = bytes.Length, digest = "sha256:" + hash });
        var json = JsonSerializer.Serialize(new { tag_name = version, body = "fixture", html_url = "https://github.com/fixture/updater/releases/tag/0.5.8-11", draft = isDraft, prerelease = isPrerelease, assets });
        return new(HttpStatusCode.OK) { Content = new StringContent(json, Encoding.UTF8, "application/json") };
    }
    if (request.RequestUri.AbsolutePath.EndsWith("SHA256SUMS"))
        return new(HttpStatusCode.OK) { Content = new StringContent($"{hash}  {sc}\n{hash} *{fdd}\n") };
    Check(request.RequestUri.AbsolutePath.EndsWith(sc), "legacy downloads SC only");
    return new(HttpStatusCode.OK) { Content = new ByteArrayContent(bytes) };
};
var service = new UpdateService();

    .GetMethod("FetchLatestAsync", BindingFlags.NonPublic | BindingFlags.Instance)!.Invoke(service, new object[] { CancellationToken.None })!;

    Check(release is { AssetName: sc, Version: version } && release.Sha256 == hash, "formal release selects SC with FDD last and verifies checksum metadata");
    var downloaded = Path.Combine(root, digest ? "digest.zip" : "sums.zip");
    await (Task)typeof(UpdateService).GetMethod("DownloadAsync", BindingFlags.NonPublic | BindingFlags.Instance)!
        .Invoke(service, new object[] { release!, downloaded, CancellationToken.None })!;
    var staged = UpdateService.ExtractPackage(downloaded, Path.Combine(root, digest ? "digest-files" : "sums-files"), version);
    var install = Path.Combine(root, digest ? "digest-install" : "sums-install");
    Directory.CreateDirectory(install);
    File.WriteAllText(Path.Combine(install, "MikuN2N.exe"), "old fixture");
    UpdateService.ReplaceFiles(staged, install);
    Check(FileVersionInfo.GetVersionInfo(Path.Combine(install, "MikuN2N.exe")).ProductVersion!.StartsWith(version), "staged EXE version and installed file");
    Check(File.ReadAllText(Path.Combine(install, "MikuN2N.exe.update-old")) == "old fixture", "old file retained");
    Console.WriteLine("PASS: immutable -10 algorithm discovers formal SC release, " + (digest ? "digest" : "SHA256SUMS") + ", downloads, checks hash, verifies EXE and replaces isolated install");

static void Check(bool condition, string message)
{
    if (!condition) throw new InvalidOperationException(message);
}

internal static class OfflineTransport
{
    internal static Func<HttpRequestMessage, HttpResponseMessage> Route = _ => throw new InvalidOperationException("unexpected HTTP");
    internal static readonly HttpClient Client = new(new Handler());
    private sealed class Handler : HttpMessageHandler
    {
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken) => Task.FromResult(Route(request));
    }
}
